Q9. Perform insertion sort. 
def insertion_sort(arr): 
    for i in range(1, len(arr)): 
        key = arr[i] 
        j = i - 1 
 
        while j >= 0 and arr[j] > key: 
            arr[j + 1] = arr[j] 
            j -= 1 
 
        arr[j + 1] = key 
 
    return arr

    if op in "+-": 
        return 1 
    if op in "*/": 
        return 2 
    if op == "^": 
        return 3 
    return 0 
 
def infix_to_prefix(expr): 
    expr = expr[::-1] 
    expr = ''.join('(' if c == ')' else ')' if c == '(' else c for c in expr) 
 
    stack = [] 
    output = [] 
 
    for c in expr: 
        if c.isalnum(): 
            output.append(c) 
        elif c == '(': 
            stack.append(c) 
        elif c == ')': 
            while stack and stack[-1] != '(': 
                output.append(stack.pop()) 
            stack.pop() 
        else: 
            while stack and stack[-1] != '(' and precedence(stack[-1]) > 
precedence(c): 
                output.append(stack.pop()) 
            stack.append(c) 
 
    while stack: 
        output.append(stack.pop()) 
 
    return ''.join(output)[::-1]

Q11. Perform selection sort. 
def selection_sort(arr): 
    n = len(arr) 
 
    for i in range(n): 
        min_index = i 
 
        for j in range(i + 1, n): 
            if arr[j] < arr[min_index]: 
                min_index = j 
 
        arr[i], arr[min_index] = arr[min_index], arr[i] 
 
    return arr 
 
arr = [64, 25, 12, 22, 11] 
print("Original:", arr) 
print("Sorted:", selection_sort(arr)) 
 
# Output: 
# Original: [64, 25, 12, 22, 11] 
# Sorted: [11, 12, 22, 25, 64]

def sort_stack(stack): 
    temp = [] 
 
    while stack: 
        x = stack.pop() 
 
        while temp and temp[-1] > x: 
            stack.append(temp.pop()) 
 
        temp.append(x)

def binary_search(arr, item): 
    low = 0 
    high = len(arr) - 1 
 
    while low <= high: 
        mid = (low + high) // 2 
 
        if arr[mid] == item: 
            return True 
        elif arr[mid] < item: 
            low = mid + 1 
        else: 
            high = mid - 1 
 
    return False

Q17. Perform merge sort. 
def merge_sort(arr): 
if len(arr) <= 1: 
return arr 
mid = len(arr) // 2 
    left = merge_sort(arr[:mid]) 
    right = merge_sort(arr[mid:]) 
 
    result = [] 
    i = j = 0 
 
    while i < len(left) and j < len(right): 
        if left[i] <= right[j]: 
            result.append(left[i]) 
            i += 1 
        else: 
            result.append(right[j]) 
            j += 1 
 
    result.extend(left[i:]) 
    result.extend(right[j:])

Q18. Evaluate a postfix expression using stack. 
def evaluate_postfix(expr): 
    stack = [] 
 
    for token in expr.split(): 
        if token.isdigit(): 
            stack.append(int(token)) 
        else: 
            b = stack.pop() 
            a = stack.pop() 
 
            if token == '+': 
                stack.append(a + b) 
            elif token == '-': 
                stack.append(a - b) 
            elif token == '*': 
                stack.append(a * b) 
            elif token == '/': 
                stack.append(a / b) 
 
    return stack.pop() 
 
expr = input("Enter postfix expression (space separated): ") 
print("Result:", evaluate_postfix(expr))

Q29. Implement a circular queue. 
class CircularQueue: 
    def __init__(self, size): 
        self.queue = [None] * size 
        self.size = size 
        self.front = -1 
        self.rear = -1 
 
    def enqueue(self, item): 
        if (self.rear + 1) % self.size == self.front: 
            print("Queue Overflow") 
            return 
 
        if self.front == -1: 
            self.front = 0 
 
        self.rear = (self.rear + 1) % self.size 
        self.queue[self.rear] = item 
 
    def dequeue(self): 
        if self.front == -1: 
            return "Queue Underflow" 
 
        item = self.queue[self.front] 
 
        if self.front == self.rear: 
            self.front = self.rear = -1 
        else: 
            self.front = (self.front + 1) % self.size 
 
        return item 
 
    def display(self): 
        if self.front == -1: 
            print("Queue is empty") 
            return 
 
        i = self.front 
        result = [] 
        while True: 
            result.append(self.queue[i]) 
            if i == self.rear: 
                break 
            i = (i + 1) % self.size 
 
        print("Queue:", result) 
 
q = CircularQueue(5) 
q.enqueue(10) 
q.enqueue(20) 
q.enqueue(30) 
q.display() 
print("Deleted:", q.dequeue()) 
q.display() 
 
# Output: 
# Queue: [10, 20, 30] 
# Deleted: 10 
# Queue: [20, 30]

    left_height = height(root.left) 
    right_height = height(root.right) 
 
    return (abs(left_height - right_height) <= 1 
            and is_balanced(root.left) 
            and is_balanced(root.right))

root.right = Node(30)

    print("Binary tree is balanced") 
else: 
    print("Binary tree is not balanced")

# Binary tree is balanced

Q42. Perform topological sorting. 
def topological_sort(graph): 
    indegree = {v: 0 for v in graph} 
 
    for v in graph: 
        for neighbor in graph[v]: 
            indegree[neighbor] += 1 
 
    queue = [v for v in graph if indegree[v] == 0] 
    result = [] 
 
    while queue: 
        v = queue.pop(0) 
        result.append(v) 
 
        for neighbor in graph[v]: 
            indegree[neighbor] -= 1 
            if indegree[neighbor] == 0: 
                queue.append(neighbor) 
 
    return result 
 
graph = { 
    0: [1, 2], 
    1: [3], 
    2: [3], 
    3: [] 
} 
 
print("Topological Order:", topological_sort(graph)) 
 
# Output: 
# Topological Order: [0, 1, 2, 3]

Q47. Perform bubble sort. 
def bubble_sort(arr): 
    n = len(arr) 
 
    for i in range(n): 
        for j in range(0, n - i - 1): 
            if arr[j] > arr[j + 1]: 
                arr[j], arr[j + 1] = arr[j + 1], arr[j] 
 
    return arr 
 
arr = [5, 1, 4, 2, 8] 
print("Original:", arr) 
print("Sorted:", bubble_sort(arr)) 
 
# Output: 
# Original: [5, 1, 4, 2, 8] 
# Sorted: [1, 2, 4, 5, 8]

correct position.

position. 
What is bubble sort? Repeatedly compares adjacent elements and swaps them when they are 
in the wrong order. 
What is quick sort? A divide-and-conquer sorting method based on a pivot. 
What is merge sort? A divide-and-conquer sort that divides the list and merges sorted parts.
